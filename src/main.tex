%-------------------------------------------------------------------------------
% main.tex — entry point. Compile from the repository root with:
%   mkdir -p build && xelatex -output-directory=build src/main.tex
% (twice, for the QR code position), or with: make build
%
% Layout/style lives in  gmresume.cls  (how the resume looks; reusable class)
% Content lives in        content.tex   (what the resume says)
% This file only picks the class settings that differ from its defaults.
%-------------------------------------------------------------------------------
\documentclass{gmresume}

% SF Pro, from this repository (not distributed with the class).
\resumeset{
  fontdir        = assets/fonts/,
  fontregular    = SF-Pro-Text-Regular.otf,
  fontbold       = SF-Pro-Text-Semibold.otf,
  fontitalic     = SF-Pro-Text-RegularItalic.otf,
  fontbolditalic = SF-Pro-Text-SemiboldItalic.otf,
  fontlight      = SF-Pro-Text-Light.otf,
}

% content.tex only declares elements (fills the header/metadata fields and the
% section buckets) and typesets nothing, so it is read in the preamble: the PDF
% metadata it declares must be set before \begin{document}.
\input{src/content.tex}

\begin{document}
% \makeresume assembles the declared elements into the laid-out document.
\makeresume
\end{document}
